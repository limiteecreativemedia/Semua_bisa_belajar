const shuffleArray = (array) => {
  const newArray = [...array];
  for (let i = newArray.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [newArray[i], newArray[j]] = [newArray[j], newArray[i]];
  }
  return newArray;
};

      <div className="fixed bottom-0 left-0 right-0 p-4 bg-white/90 backdrop-blur-md border-t flex justify-center max-w-md mx-auto z-40 shadow-[0_-10px_20px_rgba(0,0,0,0.05)]">
        <button onClick={() => setCurrentView('practice')} className="w-full bg-blue-600 hover:bg-blue-700 text-white font-extrabold py-4 px-6 rounded-2xl shadow-[0_6px_0_rgb(37,99,235)] hover:translate-y-1 hover:shadow-[0_2px_0_rgb(37,99,235)] transition-all text-lg flex justify-center items-center gap-2">
          COBA LATIHAN ACAK <ArrowRight size={20} />
        </button>
      </div>
    </div>
  );
};

// --- QUIZ ENGINE (RANDOMIZER) ---
const QuizEngine = ({ isExam, activeLevel, CONTENT, currentView, setCurrentView, userData, updateStats, addStars, unlockNextLevel }) => {
  const [questions, setQuestions] = useState([]);
  const [qIndex, setQIndex] = useState(0);
  const [selectedAns, setSelectedAns] = useState(null);
  const [status, setStatus] = useState('idle');
  const [hintLevel, setHintLevel] = useState(0);
  const [mistakes, setMistakes] = useState(0);
  const [examScore, setExamScore] = useState(0);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    setIsLoading(true);
    const pool = isExam ? CONTENT[activeLevel.id].examPool : CONTENT[activeLevel.id].practicePool;
    const shuffledPool = shuffleArray(pool);
    const selectedQuestions = shuffledPool.slice(0, 5); // Ambil 5 soal acak
    setQuestions(selectedQuestions);
    setTimeout(() => setIsLoading(false), 500);
  }, [activeLevel.id, isExam]);

  if (isLoading) {
    return (
      <div className="min-h-screen bg-gray-50 flex flex-col items-center justify-center max-w-md mx-auto text-blue-600">
        <RefreshCw size={48} className="animate-spin mb-4" />
        <h2 className="font-bold text-xl">Mengacak Soal...</h2>
      </div>
    );
  }

  if (currentView === 'examResult') {
    const percentage = Math.round((examScore / questions.length) * 100);
    const passed = percentage >= 80;
    
    return (
      <div className="min-h-screen bg-gradient-to-b from-blue-50 to-white p-6 flex flex-col justify-center max-w-md mx-auto text-center animate-fade-in">
         <div className="bg-white p-8 rounded-[2rem] shadow-2xl border-4 border-blue-100">
           <div className="text-7xl mb-6 drop-shadow-md">{passed ? '🏆' : '💪'}</div>
           <h2 className="text-3xl font-black text-gray-800 mb-3">{passed ? 'LULUS LEVEL!' : 'JANGAN MENYERAH!'}</h2>
           <p className="text-gray-600 font-bold mb-8 text-lg">
             {passed ? 'Luar biasa! Kamu semakin dekat menjadi Master Olimpiade.' : 'Nilaimu belum 80. Kamu bisa ulangi ujian, soalnya akan diacak baru!'}
           </p>
           
           <div className="bg-gray-50 rounded-3xl p-6 mb-8 border border-gray-100 shadow-inner">
             <div className="text-xs text-gray-500 font-black mb-2 tracking-widest uppercase">Skor Akhir</div>
             <div className={`text-6xl font-black mb-2 ${passed ? 'text-green-500' : 'text-orange-500'}`}>{percentage}</div>
             <div className="text-gray-600 font-bold bg-white inline-block px-4 py-2 rounded-xl shadow-sm border border-gray-100">
                Benar {examScore} dari {questions.length} soal
             </div>
           </div>

           {passed ? (
             <button 
               onClick={() => {
                 unlockNextLevel(activeLevel.id);
                 addStars(100);
                 setCurrentView('map');
               }} 
               className="w-full bg-green-500 hover:bg-green-600 text-white font-extrabold py-4 px-6 rounded-2xl shadow-[0_6px_0_rgb(34,197,94)] hover:translate-y-1 hover:shadow-[0_2px_0_rgb(34,197,94)] flex justify-center items-center gap-2 text-lg transition-all">
               {activeLevel.id === 10 ? 'SELESAIKAN GAME!' : 'KLAIM HADIAH & LANJUT'} <ArrowRight size={24}/>
             </button>
           ) : (
             <button 
               onClick={() => setCurrentView('lesson')} 
               className="w-full bg-blue-600 hover:bg-blue-700 text-white font-extrabold py-4 px-6 rounded-2xl shadow-[0_6px_0_rgb(37,99,235)] hover:translate-y-1 hover:shadow-[0_2px_0_rgb(37,99,235)] text-lg transition-all">
               PELAJARI MATERI LAGI
             </button>
           )}
         </div>
      </div>
    );
  }

  if (currentView === 'examIntro') {
    return (
      <div className="min-h-screen bg-gradient-to-b from-orange-50 to-white p-6 flex flex-col justify-center max-w-md mx-auto text-center animate-fade-in">
        <div className="bg-white p-8 rounded-[2rem] shadow-xl border-4 border-orange-200">
          <div className="bg-orange-100 w-24 h-24 mx-auto rounded-full flex items-center justify-center mb-6 shadow-md border-4 border-white">
            <Target size={48} className="text-orange-500"/>
          </div>
          <h2 className="text-2xl font-black text-gray-800 mb-3">Ujian {activeLevel.title}</h2>
          <p className="text-gray-600 font-bold mb-8 text-lg">Buktikan kemampuanmu dengan 5 soal acak tanpa bantuan Hint.</p>
          
          <div className="bg-orange-50 p-5 rounded-2xl mb-8 border border-orange-100 text-left shadow-inner">
            <p className="font-extrabold text-orange-900 text-sm mb-3 uppercase tracking-wider">Aturan Ujian:</p>
            <ul className="text-sm text-orange-800 space-y-2 font-bold">
              <li className="flex gap-2 items-start"><span className="text-orange-500">•</span> Tidak ada tombol Hint.</li>
              <li className="flex gap-2 items-start"><span className="text-orange-500">•</span> Hasil benar/salah muncul di akhir.</li>
              <li className="flex gap-2 items-start"><span className="text-orange-500">•</span> Syarat lulus: Skor minimal 80.</li>
            </ul>
          </div>

          <button onClick={() => setCurrentView('exam')} className="w-full bg-orange-500 hover:bg-orange-600 text-white font-extrabold py-4 px-6 rounded-2xl shadow-[0_6px_0_rgb(249,115,22)] hover:translate-y-1 hover:shadow-[0_2px_0_rgb(249,115,22)] text-lg flex justify-center items-center gap-2 transition-all">
            MULAI UJIAN <Zap size={20} fill="currentColor"/>
          </button>
        </div>
      </div>
    )
  }

  const q = questions[qIndex];
  if (!q) return null;

  const handleAnswerClick = (opt) => {
    if (isExam) {
      setSelectedAns(opt);
    } else {
      if (status === 'correct') return;
      
      setSelectedAns(opt);
      const isCorrect = (opt === q.answer);
      updateStats(isCorrect);

      if (isCorrect) {
        setStatus('correct');
        addStars(10);
      } else {
        setStatus('wrong');
        setMistakes(prev => prev + 1);
      }
    }
  };

  const handleNextClick = () => {
    if (isExam) {
      const isCorrect = (selectedAns === q.a);
      updateStats(isCorrect);
      if (isCorrect) setExamScore(prev => prev + 1);
    }

    if (qIndex < questions.length - 1) {
      setQIndex(prev => prev + 1);
      setSelectedAns(null);
      setStatus('idle');
      setHintLevel(0);
      setMistakes(0);
    } else {
      if (isExam) {
        setCurrentView('examResult');
      } else {
        setCurrentView('examIntro');
      }
    }
  };

  return (
    <div className="min-h-screen bg-gray-50 flex flex-col max-w-md mx-auto animate-fade-in">
      <div className="bg-white p-4 shadow-sm flex items-center justify-between border-b relative z-10">
        <button onClick={() => setCurrentView('levelIntro')} className="text-gray-400 hover:text-gray-700 p-2 bg-gray-50 rounded-full"><ArrowLeft size={20}/></button>
        <div className="font-extrabold text-gray-800 tracking-wide text-lg flex items-center gap-2">
          {isExam ? 'UJIAN' : 'LATIHAN'} 
          <span className="bg-blue-100 text-blue-700 px-3 py-1 rounded-lg text-sm">{qIndex + 1}/{questions.length}</span>
        </div>
        <div className="flex items-center gap-1.5 font-bold text-yellow-600 bg-yellow-50 px-3 py-1.5 rounded-xl border border-yellow-200 shadow-sm">
          <Star size={16} fill="currentColor"/> {userData.stars}
        </div>
      </div>

      <div className="w-full bg-gray-200 h-2.5 shadow-inner">
        <div className="bg-gradient-to-r from-blue-400 to-blue-600 h-2.5 transition-all duration-500 ease-out rounded-r-full" style={{width: `${((qIndex) / questions.length) * 100}%`}}></div>
      </div>

      <div className="p-5 flex-grow flex flex-col overflow-y-auto pb-36">
        <div className="bg-white rounded-[2rem] p-6 shadow-sm border border-gray-100 mb-6 flex-grow">
          <h3 className="text-xl font-black text-gray-800 leading-relaxed mb-6">{isExam ? q.q : q.question}</h3>
          
          <div className="space-y-4">
            {q.options.map((opt, i) => {
              let btnStyle = "bg-white border-2 border-gray-200 text-gray-700 hover:bg-blue-50 hover:border-blue-300 shadow-sm";
              let textStyle = "";
              
              if (selectedAns === opt) {
                 if (isExam) {
                   btnStyle = "bg-blue-500 text-white border-2 border-blue-600 shadow-md ring-2 ring-blue-300 ring-offset-1 transform scale-[1.02]";
                 } else {
                   if (status === 'correct') btnStyle = "bg-green-500 text-white border-2 border-green-600 shadow-md ring-2 ring-green-300 ring-offset-1 transform scale-[1.02]";
                   if (status === 'wrong') {
                     btnStyle = "bg-red-50 text-red-700 border-2 border-red-300";
                     textStyle = "line-through opacity-70";
                   }
                 }
              } 

              return (
                <button 
                  key={i} 
                  onClick={() => handleAnswerClick(opt)}
                  disabled={status === 'correct'}
                  className={`w-full p-4 rounded-2xl font-bold text-left text-lg transition-all duration-200 ${btnStyle}`}
                >
                  <span className={`inline-block w-8 text-center opacity-70 mr-2 ${selectedAns === opt && (isExam || status==='correct') ? 'text-white' : 'text-gray-400'}`}>
                    {String.fromCharCode(65 + i)}.
                  </span> 
                  <span className={textStyle}>{opt}</span>
                </button>
              )
            })}
          </div>
        </div>

        {!isExam && (
          <div className="space-y-4">
            {status === 'idle' && mistakes > 0 && hintLevel < q.hints.length && (
              <div className="animate-fade-in bg-blue-50 rounded-2xl p-5 border border-blue-200 flex items-start gap-4 shadow-sm">
                <div className="bg-blue-100 p-2 rounded-full text-blue-600 shrink-0"><Lightbulb size={24}/></div>
                <div className="flex-grow">
                  <p className="font-bold text-blue-900 mb-2">Bingung? Jangan khawatir!</p>
                  <button onClick={() => setHintLevel(prev => prev + 1)} className="w-full bg-white text-blue-600 py-2.5 rounded-xl font-extrabold shadow-sm border border-blue-200 active:scale-95 transition text-sm hover:bg-blue-50">
                    MINTA PETUNJUK
                  </button>
                </div>
              </div>
            )}
            
            {hintLevel > 0 && status !== 'correct' && (
               <div className="animate-fade-in bg-yellow-50 rounded-2xl p-5 border border-yellow-200 flex items-start gap-4 shadow-sm">
                  <span className="text-2xl mt-1 drop-shadow-sm">💡</span>
                  <div>
                    <p className="text-xs font-black text-yellow-600 uppercase tracking-widest mb-1">Petunjuk {hintLevel}</p>
                    <p className="text-sm font-bold text-yellow-900 leading-relaxed">{q.hints[hintLevel - 1]}</p>
                  </div>
               </div>
            )}

            {status === 'correct' && (
              <div className="animate-fade-in bg-green-50 rounded-2xl p-5 border border-green-200 flex flex-col shadow-sm">
                <div className="flex items-center gap-2 font-black text-green-700 mb-3 text-lg">
                  <CheckCircle size={24} className="text-green-600"/> Jawaban Tepat!
                </div>
                <p className="text-sm text-green-900 font-bold mb-5 leading-relaxed bg-green-100/50 p-3 rounded-xl border border-green-200/50">{q.explanation}</p>
                <button onClick={handleNextClick} className="w-full bg-green-500 hover:bg-green-600 text-white font-extrabold py-4 rounded-xl shadow-[0_4px_0_rgb(34,197,94)] hover:translate-y-1 hover:shadow-[0_0px_0_rgb(34,197,94)] text-lg transition-all flex justify-center items-center gap-2">
                  LANJUT <ArrowRight size={20}/>
                </button>
              </div>
            )}

            {status === 'wrong' && (
              <div className="animate-fade-in bg-orange-50 rounded-2xl p-5 border border-orange-200 flex flex-col shadow-sm">
                <div className="flex items-center gap-2 font-black text-orange-700 mb-2 text-lg">
                  <AlertCircle size={24} className="text-orange-600"/> Hampir Benar!
                </div>
                <p className="text-sm text-orange-900 font-bold mb-4">
                  {mistakes >= 2 ? "Soal ini menantang ya? Gunakan tombol petunjuk!" : "Coba cek lagi dengan teliti."}
                </p>
                <button onClick={() => {setSelectedAns(null); setStatus('idle');}} className="w-full bg-white hover:bg-orange-100 border-2 border-orange-300 text-orange-700 font-extrabold py-3.5 rounded-xl transition-colors active:scale-95 shadow-sm">
                  COBA PILIH LAGI
                </button>
              </div>
            )}
          </div>
        )}
      </div>

      {isExam && (
        <div className="fixed bottom-0 left-0 right-0 p-4 bg-white/90 backdrop-blur-md border-t flex justify-center max-w-md mx-auto z-40 shadow-[0_-10px_20px_rgba(0,0,0,0.05)]">
          <button 
            onClick={handleNextClick}
            disabled={!selectedAns}
            className={`w-full font-extrabold py-4 px-6 rounded-2xl transition-all text-lg flex justify-center items-center gap-2
              ${selectedAns 
                ? 'bg-blue-600 text-white hover:bg-blue-700 shadow-[0_6px_0_rgb(37,99,235)] hover:translate-y-1 hover:shadow-[0_2px_0_rgb(37,99,235)]' 
                : 'bg-gray-200 text-gray-400 cursor-not-allowed shadow-none'}`}
          >
            {qIndex === questions.length - 1 ? 'SELESAI UJIAN' : 'LANJUT SOAL'} <ArrowRight size={20} />
          </button>
        </div>
      )}
    </div>
  );
};

// --- NAVIGATION BAR ---
const Navigation = ({ currentView, setCurrentView }) => (
  <div className="fixed bottom-0 left-0 right-0 bg-white border-t border-gray-100 flex justify-around p-3 z-50 max-w-md mx-auto pb-safe shadow-[0_-10px_20px_rgba(0,0,0,0.03)]">
    {[
      { id: 'home', icon: Home, label: 'Home' },
      { id: 'map', icon: Map, label: 'Level' },
      { id: 'dashboard', icon: User, label: 'Profil' }
    ].map(nav => (
      <button 
        key={nav.id} 
        onClick={() => setCurrentView(nav.id)}
        className={`flex flex-col items-center p-2 rounded-xl transition-all duration-300 ${currentView === nav.id ? 'text-blue-600 transform scale-110' : 'text-gray-400 hover:text-gray-600'}`}
      >
        <nav.icon size={24} className="mb-1" strokeWidth={currentView === nav.id ? 2.5 : 2} />
        <span className="text-[10px] font-bold">{nav.label}</span>
        {currentView === nav.id && <div className="w-1.5 h-1.5 bg-blue-600 rounded-full mt-1.5 animate-pulse"></div>}
      </button>
    ))}
  </div>
);

// --- MAIN APP ---
export default function App() {
  const [currentView, setCurrentView] = useState('home'); 
  const [activeLevel, setActiveLevel] = useState(null);
  
  // Storage Key Final: Hanya Level 1 yang terbuka di awal, sisanya dikunci secara ketat.
  const [userData, setUserData] = useState(() => {
    const saved = localStorage.getItem('sbb_data_final');
    if (saved) return JSON.parse(saved);
    return {
      stars: 0,
      completedLevels: [],
      unlockedLevels: [1], 
      badges: [],
      streak: 1,
      stats: { totalQuestions: 0, correctAnswers: 0, weakTopics: [] }
    };
  });

  useEffect(() => {
    localStorage.setItem('sbb_data_final', JSON.stringify(userData));
  }, [userData]);

  const addStars = (amount) => {
    setUserData(prev => ({ ...prev, stars: prev.stars + amount }));
  };

  const unlockNextLevel = (currentLevelId) => {
    setUserData(prev => {
      const nextLevel = currentLevelId + 1;
      const newCompleted = prev.completedLevels.includes(currentLevelId) ? prev.completedLevels : [...prev.completedLevels, currentLevelId];
      const newUnlocked = prev.unlockedLevels.includes(nextLevel) ? prev.unlockedLevels : [...prev.unlockedLevels, nextLevel];
      
      let newBadges = [...prev.badges];
      const badgesMap = {
        1: "🌟 Pemula Hebat", 2: "🔢 Penakluk Angka", 3: "🍕 Master Pecahan", 4: "🧠 Detektif Logika",
        5: "📐 Raja Bangun", 6: "⚡ Master Cepat", 7: "🎲 Ahli Kombinasi", 8: "💡 Pemikir Jenius",
        9: "🔥 Penantang Elite", 10: "🏆 OLYMPIAD MASTER"
      };
      
      if (badgesMap[currentLevelId] && !newBadges.includes(badgesMap[currentLevelId])) {
        newBadges.push(badgesMap[currentLevelId]);
      }
      
      return { ...prev, completedLevels: newCompleted, unlockedLevels: newUnlocked, badges: newBadges };
    });
  };

  const updateStats = (isCorrect) => {
    setUserData(prev => ({
      ...prev,
      stats: {
        ...prev.stats,
        totalQuestions: prev.stats.totalQuestions + 1,
        correctAnswers: prev.stats.correctAnswers + (isCorrect ? 1 : 0)
      }
    }));
  };

  return (
    <div className="min-h-screen bg-gray-50 font-sans text-gray-800 selection:bg-blue-200 pb-safe">
      <style dangerouslySetInnerHTML={{__html: `
        @import url('https://fonts.googleapis.com/css2?family=Nunito:wght@400;600;700;800;900&display=swap');
        body { font-family: 'Nunito', sans-serif; }
        @keyframes fadeIn { from { opacity: 0; transform: translateY(15px); } to { opacity: 1; transform: translateY(0); } }
        .animate-fade-in { animation: fadeIn 0.4s cubic-bezier(0.16, 1, 0.3, 1) forwards; }
        .pb-safe { padding-bottom: env(safe-area-inset-bottom, 20px); }
      `}} />
      
      {currentView === 'home' && <HomeView setCurrentView={setCurrentView} />}
      {currentView === 'map' && <MapView LEVELS={LEVELS} CONTENT={CONTENT} userData={userData} setCurrentView={setCurrentView} setActiveLevel={setActiveLevel} />}
      {currentView === 'dashboard' && <DashboardView userData={userData} />}
      
      {currentView === 'levelIntro' && <LevelIntroView activeLevel={activeLevel} setCurrentView={setCurrentView} />}
      {currentView === 'lesson' && <LessonView activeLevel={activeLevel} CONTENT={CONTENT} setCurrentView={setCurrentView} />}
      
      {['practice', 'exam', 'examIntro', 'examResult'].includes(currentView) && (
        <QuizEngine 
          key={`${currentView === 'practice' ? 'practice' : 'exam'}-${activeLevel?.id}`} 
          isExam={currentView !== 'practice'} 
          activeLevel={activeLevel}
          CONTENT={CONTENT}
          currentView={currentView}
          setCurrentView={setCurrentView}
          userData={userData}
          updateStats={updateStats}
          addStars={addStars}
          unlockNextLevel={unlockNextLevel}
        />
      )}

      {['home', 'map', 'dashboard'].includes(currentView) && <Navigation currentView={currentView} setCurrentView={setCurrentView} />}
    </div>
  );
}
